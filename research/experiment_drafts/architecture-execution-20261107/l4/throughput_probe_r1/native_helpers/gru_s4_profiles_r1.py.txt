R = Path("/data2/lc-epanov_s/tabdata/runtime/architecture-execution-20261107-r1")
S1, S4 = R / "source-r1", R / "source-gru-monitor-resume-r1"
C = R / "control/gru-s4-profiles-r1"
P = R / "configs/gru-monitor-continuation-r3"
PYTHON = Path("/tmp/tabdata-lc-epanov_s-architecture-20261107-env/bin/python")
PLAN_SHA = "e5557696a25543daf4c7febb9d8f343451ce5ec5d47c857a3f2b232d19d88b5b"
S1_SHA = "7d3324230effb050ff083b939e26ad638e5df5c5b72ea70b0f8aed169997db4a"

ALLOWED = {
    "src/wnn/training/sequence_monitor_gates.py",
    "src/wnn/training/sequence_monitor_diagnostics.py",
    "src/wnn/training/sequence_config_validation.py",
    "src/wnn/training/sequence_fit_resume.py",
}


def sha(path: Path) -> str:
    """Hash actual bounded configuration/control/source bytes."""
    return hashlib.sha256(path.read_bytes()).hexdigest()


def canonical(value: object) -> str:
    """Match canonical producer identities without importing a model."""
    return hashlib.sha256(
        json.dumps(
            value,
            sort_keys=True,
            ensure_ascii=False,
            separators=(",", ":"),
            allow_nan=False,
        ).encode()
    ).hexdigest()


def pinned(path: Path, expected: str) -> dict:
    """Require a ROOT-supplied digest and a direct regular JSON file."""
    if path.is_symlink() or not path.is_file() or sha(path) != expected:
        raise ValueError("ROOT-pinned native artifact changed")
    return json.loads(path.read_bytes())


def check_sources(manifest_sha: str, receipt_sha: str) -> dict:
    """Admit only620 S4 files and the cumulative four-file amendment against frozen S1."""
    manifest = pinned(R / "source-gru-monitor-resume-r1-manifest.json", manifest_sha)
    patch = pinned(R / "source-gru-monitor-resume-r1-patch-receipt.json", receipt_sha)
    parent = pinned(R / "manifests/l4-source-r1-manifest.json", S1_SHA)
    old, new = parent["files"], manifest["files"]
    if (
        len(old) != 620
        or old.keys() != new.keys()
        or {name for name in old if old[name] != new[name]} != ALLOWED
        or manifest["parent_inventory_sha256"] != S1_SHA
        or manifest["diagnostic_parent_inventory_sha256"] != S3_SHA
        or patch["parent_inventory_sha256"] != S1_SHA
        or patch["new_inventory_sha256"] != manifest_sha
        or patch["changed_files_relative_to_S3"]
        != ["src/wnn/training/sequence_fit_resume.py"]
        or patch["changed_files"]
        != {
            name: {"parent_sha256": old[name], "new_sha256": new[name]}
            for name in ALLOWED
        }
    ):
        raise ValueError("S4 inventory/ancestry escaped the exact four-file whitelist")
    for name in new:
        relative = PurePosixPath(name)
        if relative.is_absolute() or ".." in relative.parts or "\\" in name:
            raise ValueError("unsafe source inventory path")
        for root, expected in ((S1, old[name]), (S4, new[name])):
            path = root.joinpath(*relative.parts)
            if (
                path.is_symlink()
                or not path.resolve().is_relative_to(root.resolve())
                or sha(path) != expected
            ):
                raise ValueError("frozen source file bytes changed")
    expected_python = {
        name
        for name in new
        if name.endswith(".py") and name.startswith(("src/", "scripts/"))
    }
    actual_python = {
        path.relative_to(S4).as_posix()
        for root in (S4 / "src", S4 / "scripts")
        for path in root.rglob("*.py")
    }
    if actual_python != expected_python:
        raise ValueError("S4 has extra or missing importable files")
    trainer = {
        name: value
        for name, value in new.items()
        if name.startswith("src/")
        and name.endswith(".py")
        or name == "scripts/train_sequence.py"
    }
    return {
        "manifest_sha256": manifest_sha,
        "patch_receipt_sha256": receipt_sha,
        "source_sha256": canonical(trainer),
    }


def query(arguments: list[str]) -> list[list[str]]:
    """Read GPU telemetry only; never signal or inspect a training process."""
    result = subprocess.run(
        ["nvidia-smi", *arguments, "--format=csv,noheader,nounits"],
        check=True,
        capture_output=True,
        text=True,
        timeout=10,
    )
    return [
        [field.strip() for field in line.split(",")]
        for line in result.stdout.splitlines()
        if line.strip()
    ]


def profile_entry(entry: dict, physical: dict[int, str], active: set[str]) -> dict:
    """Bind exact config/data/physical GPU and prepare one20-update canonical command."""
    seed, gpu = entry["seed"], entry["gpu_index"]
    metadata = pinned(
        R / f"runs/gru_s{seed}/fit/run_metadata.json", entry["old_metadata_sha256"]
    )
    config = pinned(P / entry["config_file"], entry["config_sha256"])
    normalized = dict(metadata["config"]) | {"checkpoint_retention": "final_only"}
    raw = json.loads(json.dumps(normalized))
    raw["model"].pop("precision_mode")
    runtime = metadata["provenance"]["runtime"]
    if (
        config != raw
        or canonical(normalized) != entry["normalized_config_sha256"]
        or normalized["seed"] != seed
        or normalized["model"]["architecture"] != "gru"
        or normalized["epochs"] != 12
        or normalized["batch_size"] != 16
        or normalized["required_world_size"] != 1
        or metadata["provenance"]["world_size"] != 1
        or runtime["visible_devices"] != 1
        or len(runtime["devices"]) != 1
    ):
        raise ValueError("profile changed the exact normalized GRU recipe")
    uuid = "GPU-" + runtime["devices"][0]["uuid"].removeprefix("GPU-")
    if physical.get(gpu) != uuid or uuid in active:
        raise ValueError(
            "selected physical GPU differs or has an active compute process"
        )
    output = R / f"runs/gru-monitor-continuation-profiles-r4/gru_s{seed}"
    if output.exists():
        raise FileExistsError("S4 engineering profile output already exists; no retry")
    env = dict(os.environ)
    for key in entry["env_unset"]:
        env.pop(key, None)
    env.update(entry["env_set"])
    env.update(
        PYTHONPATH=str(S4 / "src"),
        CUDA_VISIBLE_DEVICES=uuid,
        WNN_MONITOR_HEARTBEAT_SECONDS="5",
    )
    argv = [
        str(PYTHON),
        str(S4 / "scripts/train_sequence.py"),
        "--config",
        str(P / entry["config_file"]),
        "--data-root",
        str(R / "project"),
        "--split-manifest",
        str(R / "project/chronological_primary.json"),
        "--output",
        str(output),
        "--profile-updates",
        "20",
    ]
    return {
        "seed": seed,
        "gpu": gpu,
        "uuid": uuid,
        "output": output,
        "env": env,
        "argv": argv,
        "config_sha256": canonical(normalized),
        "data_sha256": metadata["provenance"]["data_sha256"],
    }


def run_profile(
    item: dict, binding: dict, helpers: SimpleNamespace, owner: dict
) -> None:
    """Use the accepted relay and200-second owned timeout, with no progress callback."""
    seed, output = item["seed"], item["output"]
    command = [
        str(PYTHON),
        str(R / "control/compact-receiving-r1/observed_worker.py"),
        "--",
        str(PYTHON),
        str(S4 / "scripts/observe_sequence.py"),
        "--output",
        str(output),
        "--interval",
        "10",
        "--max-seconds",
        "195",
        "--",
        *item["argv"],
    ]
    code, timed_out, _ = helpers.run(
        command,
        cwd=S4,
        env=item["env"],
        log_path=C / f"profile-{seed}.log",
        receipt_path=C / f"owned-profile-{seed}.json",
        owner={"controller": owner, "cell_id": f"gru_s{seed}"},
        stage=1,
        trial_number=item["gpu"],
        mode="engineering_profile",
        gpu_key=str(item["gpu"]),
        output_dir=output,
        timeout_seconds=200,
    )
    result = {
        "exit_code": code,
        "timed_out": timed_out,
        "seed": seed,
        "physical_uuid": item["uuid"],
        "scientific_fits": 0,
        "approved": False,
    }
    helpers.write(C / f"result-{seed}.json", result, replace=False)
    if code or timed_out:
        raise RuntimeError("bounded S4 profile failed; no automatic retry")
    receipt = json.loads((output / "profile_receipt.json").read_bytes())
    identity = receipt["identity"]
    devices = identity["runtime"]["devices"]
    if (
        receipt["status"] != "profile_complete"
        or receipt["updates"] != 20
        or receipt["errors"]
        or receipt["scientific_checkpoint_written"] is not False
        or identity["world_size"] != 1
        or identity["source_sha256"] != binding["source_sha256"]
        or identity["config_sha256"] != item["config_sha256"]
        or identity["data_sha256"] != item["data_sha256"]
        or len(devices) != 1
        or "GPU-" + devices[0]["uuid"].removeprefix("GPU-") != item["uuid"]
    ):
        raise ValueError(
            "native S4 profile receipt differs from exact engineering allocation"
        )


def main() -> None:
    """Prepare/run just two finite S4 profiles; never pause, stop, resume or authorize a fit."""
    parser = argparse.ArgumentParser()
    parser.add_argument("--source-manifest-sha256", required=True)
    parser.add_argument("--patch-receipt-sha256", required=True)
    parser.add_argument("--detach", action="store_true")
    args = parser.parse_args()
    if any(
        not re.fullmatch("[0-9a-f]{64}", value)
        for value in (args.source_manifest_sha256, args.patch_receipt_sha256)
    ):
        parser.error(
            "ROOT must provide the actual native S4 manifest and patch-receipt SHA256"
        )
    sys.path.insert(0, str(S1 / "src"))
    from wnn.multistage_search import atomic_write_json, file_lock
    from wnn.multistage_search_owned_command import run_owned_command
    from wnn.multistage_search_runtime import (
        current_process_owner,
        launch_detached_worker,
    )

    C.mkdir(parents=True, exist_ok=True)
    if args.detach:
        command = [str(PYTHON), str(Path(__file__).resolve()), *sys.argv[1:]]
        command.remove("--detach")
        owner = launch_detached_worker(
            command,
            cwd=R,
            env=dict(os.environ),
            receipt_path=C / "owner.json",
            metadata={"scientific_fits": 0, "profiles": 2},
        )
        sys.stdout.write(json.dumps(owner) + "\n")
        return
    with file_lock(C / "profiles.lock", blocking=False):
        binding = check_sources(args.source_manifest_sha256, args.patch_receipt_sha256)
        plan = pinned(P / "CONTINUATION_PROFILE_PLAN_R3.json", PLAN_SHA)
        if [(entry["seed"], entry["gpu_index"]) for entry in plan["cells"]] != [
            (101001, 0),
            (101002, 1),
        ]:
            raise ValueError("only the original two GRU seeds/GPU0,1 may be profiled")
        physical = {int(row[0]): row[1] for row in query(["--query-gpu=index,uuid"])}
        active = {row[0] for row in query(["--query-compute-apps=gpu_uuid,pid"])}
        items = [profile_entry(entry, physical, active) for entry in plan["cells"]]
        atomic_write_json(
            C / "authority.json",
            {
                "source_binding": binding,
                "plan_sha256": PLAN_SHA,
                "profile_cap": 2,
                "profile_updates": 20,
                "timeout_seconds": 200,
                "scientific_fits": 0,
                "approved": False,
            },
            replace=False,
        )
        helpers = SimpleNamespace(write=atomic_write_json, run=run_owned_command)
        owner = current_process_owner()
        with ThreadPoolExecutor(max_workers=2) as pool:
            futures = [
                pool.submit(run_profile, item, binding, helpers, owner)
                for item in items
            ]
            for future in futures:
                future.result()


if __name__ == "__main__":
    main()
